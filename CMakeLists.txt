cmake_minimum_required(VERSION 3.20)
project(luaspdlog VERSION 0.1.0 LANGUAGES C CXX)
option(LUASPDLOG_STATIC "构建供移动端或嵌入宿主使用的静态库" OFF)
option(LUASPDLOG_TESTS "注册 Lua 集成测试" ON)
option(LUASPDLOG_SANITIZE "启用地址及未定义行为检测" OFF)
set(LUASPDLOG_LUA_SOURCE_DIR "" CACHE PATH "官方 Lua 源码目录；留空则查找系统 Lua")
set(LUASPDLOG_LUA_ABI "54" CACHE STRING "Lua ABI，如 51、54、55")
set(LUASPDLOG_LUA_EXECUTABLE "" CACHE FILEPATH "用于测试的 Lua 解释器")
if(LUASPDLOG_LUA_SOURCE_DIR)
  include(cmake/LuaRuntime.cmake)
else()
  find_package(Lua REQUIRED)
endif()
find_package(Threads REQUIRED)
if(LUASPDLOG_STATIC)
  add_library(luaspdlog STATIC src/luaspdlog.cpp)
  target_compile_definitions(luaspdlog PUBLIC LUASPDLOG_STATIC)
else()
  add_library(luaspdlog MODULE src/luaspdlog.cpp)
endif()
target_compile_features(luaspdlog PRIVATE cxx_std_17)
target_include_directories(luaspdlog PRIVATE "${LUA_INCLUDE_DIR}" third_party/spdlog/include PUBLIC include)
target_link_libraries(luaspdlog PRIVATE Threads::Threads)
if(WIN32)
  target_compile_definitions(luaspdlog PRIVATE SPDLOG_WCHAR_FILENAMES)
  target_link_libraries(luaspdlog PRIVATE ${LUA_LIBRARIES})
elseif(APPLE AND NOT LUASPDLOG_STATIC)
  target_link_options(luaspdlog PRIVATE "-undefined" "dynamic_lookup")
endif()
if(LUASPDLOG_STATIC AND TARGET lua_runtime)
  add_executable(embed_check tests/embed.c)
  target_include_directories(embed_check PRIVATE "${LUA_INCLUDE_DIR}")
  target_link_libraries(embed_check PRIVATE luaspdlog lua_runtime Threads::Threads)
  set_target_properties(embed_check PROPERTIES LINKER_LANGUAGE CXX)
endif()
if(MSVC)
  target_compile_options(luaspdlog PRIVATE /W4 /utf-8 /EHsc)
else()
  target_compile_options(luaspdlog PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(LUASPDLOG_SANITIZE AND NOT MSVC)
  target_compile_options(luaspdlog PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(luaspdlog PRIVATE -fsanitize=address,undefined)
  if(TARGET lua_test_runtime)
    foreach(t lua_runtime lua_test_runtime)
      target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${t} PRIVATE -fsanitize=address,undefined)
    endforeach()
  endif()
endif()
set_target_properties(luaspdlog PROPERTIES PREFIX "" CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
if(NOT WIN32 AND NOT LUASPDLOG_STATIC)
  set_target_properties(luaspdlog PROPERTIES SUFFIX ".so")
endif()
if(LUASPDLOG_STATIC)
  install(TARGETS luaspdlog ARCHIVE DESTINATION lib)
else()
  install(TARGETS luaspdlog LIBRARY DESTINATION lib RUNTIME DESTINATION lib)
endif()
install(FILES include/luaspdlog.h DESTINATION include)
install(FILES LICENSE README.md docs/API.md DESTINATION share/luaspdlog)
install(FILES third_party/spdlog/LICENSE DESTINATION share/luaspdlog/licenses RENAME spdlog.txt)
install(FILES third_party/spdlog/include/spdlog/fmt/bundled/LICENSE DESTINATION share/luaspdlog/licenses RENAME fmt.txt)
install(DIRECTORY examples tests DESTINATION share/luaspdlog FILES_MATCHING PATTERN "*.lua")
if(LUASPDLOG_TESTS AND NOT CMAKE_CROSSCOMPILING AND NOT LUASPDLOG_STATIC)
  if(NOT LUASPDLOG_LUA_EXECUTABLE)
    find_program(LUASPDLOG_LUA_EXECUTABLE NAMES lua lua5.5 lua5.4 lua5.3 lua5.2 lua5.1 luajit REQUIRED)
  endif()
  enable_testing()
  add_test(NAME integration COMMAND "${LUASPDLOG_LUA_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test.lua" "$<TARGET_FILE_DIR:luaspdlog>")
  add_executable(multistate tests/multistate.cpp)
  target_include_directories(multistate PRIVATE "${LUA_INCLUDE_DIR}")
  target_link_libraries(multistate PRIVATE ${LUA_LIBRARIES} Threads::Threads)
  set_target_properties(multistate PROPERTIES ENABLE_EXPORTS ON)
  if(LUASPDLOG_SANITIZE AND NOT MSVC)
    target_compile_options(multistate PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(multistate PRIVATE -fsanitize=address,undefined)
  endif()
  add_test(NAME multistate COMMAND multistate "$<TARGET_FILE:luaspdlog>")
endif()
